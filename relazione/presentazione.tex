\begin{frame}{Il Dataset IBM Telco \& la Pipeline di Preprocessing}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \setlength{\itemsep}{4pt}
                \item \textbf{Descrizione del Dataset}: 7.043 clienti e 21 variabili relative a dati demografici, contrattuali e finanziari.
                
                \item \textbf{Data Cleaning}: Identificati e corretti 11 valori vuoti in \texttt{TotalCharges} relativi a nuovi clienti con \textit{tenure}=0 (imputati a \texttt{0.0}).
                
                \item \textbf{Sbilanciamento del Target}: Tasso di churn pari al \textbf{26.54\%}. Un classificatore banale otterrebbe il 73.4\% di accuratezza ma \textbf{Recall pari a 0}, risultando inutile per l'azienda.
                
                \item \textbf{Feature Scaling \& Encoding}: Applicato \textbf{StandardScaler} sulle variabili continue e \textbf{One-Hot Encoding} sulle categoriali.
            \end{itemize}
        \end{column}

% SLIDE 4: CLUSTERING K-MEANS & SELEZIONE K
% ==========================================
\begin{frame}{Apprendimento Non Supervisionato: Clustering K-Means}
    \begin{columns}[c]
        \begin{column}{0.52\textwidth}
            \begin{itemize}
                \setlength{\itemsep}{4pt}
                \item \textbf{Obiettivo}: Raggruppare i clienti in segmenti omogenei senza utilizzare le etichette di churn, scoprendo la struttura latente dei dati.
                
                \item \textbf{Metodo del Gomito (Inerzia)}: Valuta la compattezza interna dei gruppi. La curva dell'inerzia mostra un chiaro punto di gomito in corrispondenza di \textbf{$K=3$}.
                
                \item \textbf{Silhouette Score}: Misura la qualità della separazione tra i cluster:
                \begin{itemize}
                    \item $K=2$: $0.295$ (distinzione banale con/senza servizio internet).
                    \item \textbf{$K=3$}: $\mathbf{0.286}$ (\textbf{ottimo equilibrio tra qualità geometrica e interpretabilità commerciale}).
                    \item $K \ge 4$: $s < 0.228$ (sovrapposizione ed over-clustering).
                \end{itemize}
            \end{itemize}
        \end{column}

        \begin{column}{0.46\textwidth}
            \centering
            \includegraphics[width=\textwidth,height=0.64\textheight,keepaspectratio]{images/clustering_scelta_k.png}
            \captionof{figure}{Valutazione del numero ottimale di cluster via Gomito e Silhouette.}
        \end{column}
    \end{columns}
\end{frame}

% SLIDE 7: ARCHITETTURA MODELLI SUPERVISIONATI
% ==========================================
\begin{frame}{Apprendimento Supervisionato: I Tre Classificatori}
    \begin{itemize}
        \item \textbf{Setup Sperimentale}: \textbf{Stratified Train-Test Split} (80\% Train, 20\% Test - 1.409 clienti nel test set) per preservare l'esatta percentuale del target (26.54\%).
    \end{itemize}

    \vspace{2mm}
    \begin{columns}[T]
        \begin{column}{0.32\textwidth}
            \textbf{1. Regressione Logistica}
            \vspace{2mm}
            \begin{itemize}
                \small
                \item Baseline lineare probabilistica.
                \item Trasforma gli input in probabilità tramite funzione Sigmoide.
                \item Modello veloce, stabile ed interpretabile.
            \end{itemize}
        \end{column}

        \begin{column}{0.32\textwidth}
            \textbf{2. Decision Tree (CART)}
            \vspace{2mm}
            \begin{itemize}
                \small
                \item Modello White-Box trasparente.
                \item Scelta degli split tramite l'Indice di Gini.
                \item Regolarizzato a \texttt{max\_depth=5} contro l'overfitting.
            \end{itemize}
        \end{column}

        \begin{column}{0.32\textwidth}
            \textbf{3. MLP Neural Network}
            \vspace{2mm}
            \begin{itemize}
                \small
                \item Architettura Deep Learning.
                \item 2 strati nascosti ($64 \to 32$ neuroni) con attivazione ReLU.
                \item Ottimizzatore Adam, regolarizzazione L2 ed Early Stopping.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

% SLIDE 10: DISCUSSIONE CRITICA & CONCLUSIONI
% ==========================================
\begin{frame}{Discussione Critica, Conclusioni \& Sviluppi Futuri}
    \begin{itemize}
        \setlength{\itemsep}{5pt}
        \item \textbf{Unsupervised Feature Engineering}: La variabile di cluster fornisce alla rete neurale un'informazione aggregata pre-calcolata che guida l'ottimizzatore Adam fuori da minimi locali, aumentando la sensibilità verso i clienti in churn.
        
        \item \textbf{Priorità di Business}: In ambito Telco la \textbf{Recall} (Sensitivity) è la metrica prioritaria, poiché il costo aziendale di un cliente perso non identificato è nettamente superiore al costo di un'offerta promozionale.
        
        \item \textbf{Sintesi dei Risultati}: K-Means ($K=3$) isola con precisione il segmento a rischio (Cluster 0 al 43.2% churn); la Regressione Logistica offre la massima capacità predittiva; il Decision Tree garantisce completa interpretabilità.
        
        \item \textbf{Sviluppi Futuri}: Bilanciamento del target tramite \textbf{SMOTE}, modelli di Ensemble Learning (\textbf{Random Forest}, \textbf{XGBoost}) e Cost-Sensitive Learning.
    \end{itemize}
\end{frame}
