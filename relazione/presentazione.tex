            \vspace{1mm}

            \footnotesize
            \begin{itemize}
                \footnotesize
                \setlength{\itemsep}{1.8pt}
                \item \textbf{Performance (P)}: Valore netto di fidelizzazione (ridurre il churn contenendo i costi rispetto al CAC).
                \item \textbf{Environment (E)}: Clientela Telco: \textit{parzialmente osservabile}, \textit{nondeterministico / stocastico}, \textit{multi-agente}, \textit{continuo}, \textit{sconosciuto} (appreso dai dati); \textit{sequenziale}/\textit{dinamico} nelle campagne (\textit{episodico}/\textit{statico} su snapshot batch).
                \item \textbf{Actuators (A)}: Offerte mirate, sconti e supporto prioritario.
                \item \textbf{Sensors (S)}: Log CRM (fatturazione, \textit{tenure}, contratti, servizi).
            \end{itemize}

Scegliamo l'Agire Razionalmente: l'agente seleziona azioni che massimizzano il valore atteso della misura di prestazione P sotto incertezza (senza onniscienza), usando una funzione di utilità per bilanciare churn evitato e costi delle offerte e apprendendo dai dati storici. Nel PEAS distinguiamo P (esito economico esterno nell'ambiente), E (parzialmente osservabile, nondeterministico/stocastico, multi-agente, continuo e sconosciuto a priori; sequenziale e dinamico nel ciclo reale, episodico e statico nella singola predizione batch), A (offerte e supporto) e S (log CRM).}

\begin{frame}{Formulazione dei Task ML e Approccio Probabilistico}
    \centering
    \colorbox{LightBg}{\parbox{0.96\textwidth}{\centering
        \small \color{PrimaryNavy}\textbf{Dal PEAS al Machine Learning (Lezione 03)}: Sistema \textit{Batch} e \textit{Model-Based} che integra apprendimento Supervisionato e Non Supervisionato.
    }}
    \vspace{1.8mm}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Definizione dei Due Task di Progetto}
            \vspace{1mm}

            \footnotesize
            \textbf{1. Task Supervisionato} (T.\ Mitchell $\langle T, E, P \rangle$):
            \begin{itemize}
                \footnotesize
                \setlength{\itemsep}{1.5pt}
                \item \textbf{Task ($T$)}: Classificazione binaria probabilistica di \texttt{Churn}.
                \item \textbf{Experience ($E$)}: $5.634$ istanze di training etichettate (su $7.043$).
                \item \textbf{Performance ($P$)}: \textit{Recall}, \textit{Precision}, \textit{F1-Score}, \textit{AUC-ROC}.
            \end{itemize}
            \vspace{1.5mm}
            \textbf{2. Task Non Supervisionato} (Esplorativo):
            \begin{itemize}
                \footnotesize
                \setlength{\itemsep}{1.5pt}
                \item Clustering \textbf{K-Means} (30 feature senza \texttt{Churn}, valutato con Inerzia WCSS e \textit{Silhouette}) + proiezione \textbf{PCA} 2D, verificandone l'effetto come feature sui classificatori.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Approccio Probabilistico Discriminativo}
            \vspace{1mm}

            \footnotesize
            \textit{Programmazione Tradizionale vs.\ Machine Learning:}
            \vspace{1mm}
            \begin{itemize}
                \footnotesize
                \setlength{\itemsep}{2pt}
                \item \textbf{Limiti delle Regole Manuali}: Regole booleane (\textit{if-then}) su $30$ feature sono fragili, non scalano ai cambi di mercato e non quantificano il rischio.
                \item \textbf{Stima Diretta $P(\text{Churn}=1 \mid \mathbf{x})$}: I modelli discriminativi (\texttt{LogReg}, \texttt{CART}, \texttt{MLP}) apprendono uno score continuo in $[0,1]$ per fare \textit{ranking} dei clienti e tarare la soglia decisionale in funzione del CAC.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{2mm}
    \centering
    \colorbox{LightBg}{\parbox{0.96\textwidth}{\centering
        \small \color{DarkText} \textbf{Nota Teorica}: $P$ nel PEAS misura l'impatto economico nell'ambiente; $P$ in Mitchell misura la qualità predittiva del modello.
    }}

    \note{Per mappare i Sensori S negli Attuatori A adottiamo un sistema ML Batch e Model-Based (Lezione 03) articolato su due task: 1) Task supervisionato secondo Mitchell <T, E, P> per stimare la probabilità di Churn con metriche robuste allo sbilanciamento; 2) Task non supervisionato (K-Means + PCA) per scoprire segmenti latenti senza usare l'etichetta Churn e verificarne l'impatto come feature. Rispetto alle regole if-then tradizionali, i classificatori probabilistici discriminativi offrono uno score continuo per ordinare il rischio e tarare la soglia decisionale.}
\end{frame}

% SLIDE 5: CLUSTERING K-MEANS & SELEZIONE K
% ==========================================
\begin{frame}{Apprendimento Non Supervisionato: Clustering K-Means}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \textbf{Scopo del Clustering e Selezione di $K$}
            \vspace{1.5mm}
            \begin{itemize}
                \small
                \setlength{\itemsep}{2.4pt}
                \item \textbf{Obiettivo}: Segmentare i $5.634$ clienti del Training Set (30 feature) senza usare l'etichetta \texttt{Churn}.
                \item \textbf{Scelta di K-Means (vs DBSCAN)}: Centroidi medi interpretabili, scalabilità lineare e \texttt{.predict()} \textit{out-of-sample} sul Test Set senza leakage.
                \item \textbf{Confronto per $K \in \{2, 3, 4, 5, 6\}$}:
                \begin{itemize}
                    \small
                    \setlength{\itemsep}{2pt}
                    \item \textbf{$K=2$} ($s=0.293$): Divisione banale (solo presenza/assenza internet).
                    \item \textbf{$K=3$ (Scelto)}: Punto di gomito dell'Inerzia WCSS ($33.485 \rightarrow 24.642$) con Silhouette pressoché invariata ($s=0.282$).
                    \item \textbf{$K \ge 4$}: Netto calo della Silhouette ($0.230 \rightarrow 0.185$) per frammentazione.
                \end{itemize}
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\textwidth,height=0.42\textheight,keepaspectratio]{images/clustering_scelta_k.png}

            \vspace{1.5mm}
            \colorbox{LightBg}{\parbox{\textwidth}{
                \small \color{DarkText}
                \textbf{Lettura delle due metriche}:
                \begin{itemize}
                    \footnotesize
                    \setlength{\itemsep}{2pt}
                    \item \textbf{Inerzia WCSS (Blu)}: Somma delle distanze quadratiche dai centroidi; mostra il gomito a \textbf{$K=3$}.
                    \item \textbf{Silhouette (Arancione)}: Combina coesione intra-cluster ($a_i$) e separazione inter-cluster ($b_i$). In 30D misto $s=0.282$ indica separazione moderata ma utile.
                \end{itemize}
            }}
        \end{column}
    \end{columns}

    \note{Applichiamo K-Means sul solo Training Set standardizzato (5.634 istanze, 30 feature) senza data leakage. Per scegliere K combiniamo Inerzia WCSS (compattezza interna, metodo del gomito) e Silhouette Score, che bilancia coesione intra-cluster a_i e separazione b_i. K=2 ha score 0.293 ma separa solo chi ha internet da chi no. K=3 coincide con il gomito dell'Inerzia mantenendo s=0.282 (valore moderato ma fisiologico in 30 dimensioni con feature binarie one-hot), mentre per K>=4 la Silhouette crolla.}
\end{frame}
