                \textbf{Prestazione (P) -- metriche per dati sbilanciati:}\\
                -- \textbf{Recall (Sensibilità):} \% di clienti in fuga intercettati.\\
                -- \textbf{Precision:} tra chi segnaliamo, quanti abbandonano davvero.\\
                -- \textbf{F1:} media armonica (dà più peso al valore più basso).\\
                -- \textbf{AUC-ROC:} bravura nel separare le 2 classi (0,5 caso -- 1 perfetto).
            }\\[2mm]

            \card{CardGreen}{\linewidth}{\scriptsize
                \textbf{\color{AccentGreen}\small 2. Compito Non Supervisionato}\\[2pt]
                Raggruppa i clienti simili con \textbf{K-Means} senza mai guardare chi ha abbandonato.\\[1.5pt]
                Li visualizza in 2D con la \textbf{PCA} (riassume 30 variabili in 2 assi) e verifica se i gruppi aiutano le previsioni.
            }
        \end{column}

        \begin{column}{0.47\textwidth}
            \raggedright
            \textbf{\color{PrimaryNavy} Perché Machine Learning e non regole fisse?}
            \vspace{1.8mm}

            \begin{itemize}
                \setlength{\itemsep}{5pt}
                \small
                \item Con \textbf{30 caratteristiche}, scrivere regole manuali è fragile e non si adatta ai cambi di mercato.
                \item Una regola fissa risponde solo \textbf{sì o no}.
                \item Il Machine Learning stima una \textbf{probabilità di abbandono da 0 a 1}.
                \item Così ordiniamo i clienti dal più a rischio al meno a rischio e scegliamo da quale soglia intervenire.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

% SLIDE 4: DATI, PREPROCESSING E BOXPLOT SPIEGATO
% ==========================================
\begin{frame}{I Dati, il Preprocessing e il Boxplot}
    \begin{columns}[c, onlytextwidth]
        \begin{column}{0.50\textwidth}
            \raggedright
            \textbf{\color{PrimaryNavy}\small Dataset IBM Telco: 7.043 clienti (26,5\% churn)}
            \vspace{1.2mm}

            \begin{itemize}
                \setlength{\itemsep}{2.4pt}
                \scriptsize
                \item \textbf{Pulizia:} tolto il codice cliente (non aiuta a generalizzare); 11 spese vuote messe a \textbf{0 euro} (clienti nuovi a 0 mesi).
                \item \textbf{Split stratificato 80/20 (Zero Data Leakage):} dividiamo \textbf{prima} di tutto in 80\% \textbf{training} (5.634) e 20\% \textbf{test} (1.409 nascosti). \textbf{Stratificato} = stesso 26,5\% di abbandoni in entrambi; \textbf{zero leakage} = niente dati del test nel training.
                \item \textbf{Standardizzazione:} i mesi arrivano a 72, la spesa supera 8.000\,\texteuro; a ogni valore \textbf{togliamo la media e dividiamo per la deviazione standard} (solo sul training) per avere lo stesso metro (0 = media, $+$ sopra, $-$ sotto).
                \item \textbf{One-Hot Encoding:} trasforma le \textbf{16 variabili testuali} in \textbf{27 colonne 0/1} (togliendo 1 categoria ridondante per variabile) $\rightarrow$ \textbf{30 variabili finali}.
                \item \textbf{Sbilanciamento:} dire sempre ``resta'' darebbe 73,5\% di accuratezza ma \textbf{Recall 0\%}.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth, height=0.40\textheight, keepaspectratio]{images/boxplot_permanenza_costi.png}

            \vspace{1.5mm}
            \card{CardRed}{\linewidth}{\scriptsize
                \textbf{\color{AccentRed}\small Il Boxplot}\\[2pt]
                -- \textbf{\color{AccentGreen} Verde} = Rimasti \quad|\quad \textbf{\color{AccentRed} Rosso} = Abbandoni.\\[1.5pt]
                -- \textbf{Linea nera} = \textbf{mediana} (dati in ordine dal piccolo al grande: il valore centrale, o media dei 2 centrali se pari).\\
                -- \textbf{Scatola} = 50\% centrale (dal 25\textdegree{} al 75\textdegree{} percentile); \textbf{baffi/pallini} = estremi e casi anomali (\textbf{outlier}).\\[1.5pt]
                -- \textbf{Esito:} chi abbandona resta poco (\textbf{10 mesi} vs \textbf{38}) e paga di più (\textbf{80\,\texteuro} vs \textbf{64\,\texteuro}).
            }
        \end{column}
    \end{columns}
\end{frame}

% SLIDE 7: I TRE MODELLI A CONFRONTO
% ==========================================
\begin{frame}{Come Prevediamo l'Abbandono: I Tre Modelli}
    \card{CardBlue}{\textwidth}{\footnotesize
        \textbf{\color{PrimaryNavy} Esperimento pulito (Zero Leakage):} dopo lo split 80/20 confrontiamo \textbf{3 modelli} sia sui dati \textbf{Base (30 variabili)} sia aggiungendo \textbf{2 colonne 0/1 del Gruppo K-Means (32 variabili)}.
    }

    \vspace{2.6mm}
    \begin{columns}[T, onlytextwidth]
        \begin{column}{0.315\textwidth}
            \card{CardBlue}{\linewidth}{\scriptsize
                \textbf{\color{PrimaryNavy}\small 1. Regressione Logistica}\\[3pt]
                Dà un \textbf{peso} a ogni dato del cliente (es. Fibra alza il rischio, tanti mesi lo abbassano), li somma e passa il totale nella curva \textbf{sigmoide} per ottenere una \textbf{probabilità da 0\% a 100\%}.\\[3pt]
                Semplice, veloce e stabile: è il nostro \textbf{punto di riferimento}.
            }
        \end{column}

        \begin{column}{0.315\textwidth}
            \card{CardGreen}{\linewidth}{\scriptsize
                \textbf{\color{AccentGreen}\small 2. Albero CART}\\[3pt]
                Fa una \textbf{catena di domande sì/no}: \textbf{sceglie in automatico} (metodo \textbf{greedy}) la variabile e la soglia che abbassano di più l'\textbf{indice di Gini}, separando meglio le classi.\\[3pt]
                Fermato a max \textbf{5 livelli} contro l'overfitting; addestrato \textbf{senza standardizzazione} per leggere subito \textbf{mesi ed euro veri}.
            }
        \end{column}

        \begin{column}{0.315\textwidth}
            \card{CardAmber}{\linewidth}{\scriptsize
                \textbf{\color{AccentAmber}\small 3. Rete Neurale MLP}\\[3pt]
                Una squadra di piccoli valutatori (\textbf{neuroni}) in \textbf{2 strati}: \textbf{64} combinano i dati del cliente, poi \textbf{32} li riassumono nella probabilità finale.\\[2pt]
                -- \textbf{ReLU:} azzera i segnali negativi e passa solo quelli utili.\\
                -- \textbf{Adam:} corregge i pesi a ogni errore nello studio.\\
                -- Contro l'\textbf{overfitting} (imparare a memoria): \textbf{L2} (pesi piccoli) + \textbf{early stopping} (si ferma appena non migliora).
            }
        \end{column}
    \end{columns}

    \vspace{2.8mm}
    \card{CardPurple}{\textwidth}{\footnotesize
        \textbf{\color{AccentPurple} Scopo del confronto:} capire quale modello prevede meglio sui clienti nuovi e se aggiungere il gruppo K-Means porta un vero vantaggio.
    }
\end{frame}

            }\\[1.8mm]

            \card{CardAmber}{\linewidth}{\scriptsize
                \textbf{\color{AccentAmber}\small 2. Il Gruppo K-Means (barre rosse) aiuta?}\\[1.5pt]
                Quasi no: sale di pochissimo solo nella Rete Neurale; cala un filo sulla Logistica (ridondanza: informazioni già presenti); identico sull'Albero.
            }\\[1.8mm]

            \card{CardRed}{\linewidth}{\scriptsize
                \textbf{\color{AccentRed}\small 3. Come intercettare più clienti in fuga?}\\[1.5pt]
                Abbassando l'allarme al \textbf{35\%} (verificato con \textbf{5-fold cross-validation} sul solo training: 4 fette studiano e 1 verifica a turno) o con \textbf{pesi bilanciati} (errore su abbandono pesa $\sim$3 volte di più), la Recall sale al \textbf{70\%--78\%} (F1 \textbf{0,614}).
            }
        \end{column}
    \end{columns}
\end{frame}

                -- \textbf{Etica e affidabilità:} garantire \textbf{privacy}, \textbf{trasparenza} sulle decisioni e \textbf{fairness} (evitare discriminazioni su età o genere).\\[2.2pt]
                -- \textbf{Sviluppo futuro -- Random Forest:} invece di un solo albero (sensibile a piccole variazioni dei dati), addestra \textbf{tanti alberi su campioni e variabili casuali diverse} che poi \textbf{votano a maggioranza}: riduce gli errori e l'\textbf{overfitting}.
            }
