\begin{frame}{Inquadramento Teorico: Epistemologia e Agente PEAS}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Epistemologia dell'IA (Russell \& Norvig)}
            \vspace{1.5mm}
            
            \small
            L'Intelligenza Artificiale si struttura su due assi portanti: \textit{Pensare} vs. \textit{Agire} e \textit{Umanamente} vs. \textit{Razionalmente}.
            
            Il presente progetto opera nel quadrante dell'\textbf{Agire Razionalmente}: l'obiettivo non è emulare il pensiero umano, ma massimizzare l'utilità attesa di retention aziendale.

            \vspace{2mm}
            \centering
            \renewcommand{\arraystretch}{1.2}
            \setlength{\arrayrulewidth}{0.8pt}
            \arrayrulecolor{PrimaryNavy}
            \tiny
            \begin{tabular}{|c|c|}
                \hline
                \textbf{Pensare Umanamente} & \textbf{Pensare Razionalmente} \\
                (Modellazione Cognitiva) & (Leggi del Pensiero) \\
                \hline
                \textbf{Agire Umanamente} & \cellcolor{SecondaryBlue!20}\textbf{Agire Razionalmente} \\
                (Test di Turing) & \cellcolor{SecondaryBlue!20}\textbf{(L'Agente Razionale)} \\
                \hline
            \end{tabular}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Modello dell'Agente Razionale (PEAS)}
            \vspace{1.5mm}
            
            \small
            Il modello \textbf{PEAS} (\textit{Performance, Environment, Actuators, Sensors}) formalizza l'agente razionale definendo il suo ambiente e le sue capacità:
            
            \begin{itemize}
                \small
                \setlength{\itemsep}{2.5pt}
                \item \textbf{Performance (P)}: Massimizzazione della Recall per la retention dei clienti.
                \item \textbf{Environment (E)}: CRM Telco stocastico, parzialmente osservabile e sequenziale.
                \item \textbf{Actuators (A)}: Offerte promozionali, sconti mirati e supporto tecnico.
                \item \textbf{Sensors (S)}: Registri di fatturazione, tenure e tipo di abbonamento.
            \end{itemize}
            
            \vspace{1mm}
            \centering
            \begin{tikzpicture}[scale=0.65, every node/.style={transform shape}, node distance=0.7cm, auto, >=stealth]

A destra vediamo la modellizzazione PEAS dell'agente: Performance per massimizzare la Recall, Environment stocastico e parzialmente osservabile, Actuators per campagne promozionali e supporto, e Sensors per dati di fatturazione e abbonamento.}

% SLIDE 3: FONDAMENTI DI ML & APPROCCIO PROBABILISTICO
% ==========================================
\begin{frame}{Fondamenti di Machine Learning e Approccio Probabilistico}
    \centering
    \colorbox{LightBg}{\parbox{0.95\textwidth}{\centering
        \small \color{PrimaryNavy}\textbf{Raccordo col PEAS}: L'Agente impara a mappare i dati dei \textbf{Sensori (S)} in azioni degli \textbf{Attuatori (A)} analizzando i dati del CRM tramite il Machine Learning.
    }}
    \vspace{2mm}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Inquadramento del Machine Learning}
            \vspace{1mm}
            
            \small
            L'agente impara dai dati passati ($E$) per svolgere il compito ($T$) valutando le prestazioni ($P$):
            
            \begin{itemize}
                \small
                \setlength{\itemsep}{2.5pt}
                \item \textbf{Task (T)}: Prevedere chi abbandonerà l'azienda (Churn).
                \item \textbf{Experience (E)}: 7.043 dati storici di clienti IBM Telco.
                \item \textbf{Performance (P)}: Misurare Recall ed AUC-ROC.
            \end{itemize}
            
            \vspace{1.5mm}
            \centering
            \small \color{PrimaryNavy} \textbf{T} $\leftarrow$ \textbf{E} $\rightarrow$ \textbf{P}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Perché usare la Probabilità?}
            \vspace{1mm}
            
            \small
            \textit{Perché non usare semplici regole fisse (SE... ALLORA...)?}
            
            \vspace{1mm}
            \begin{itemize}
                \small
                \setlength{\itemsep}{2.5pt}
                \item \textbf{Regole Fisse (Logica Booleana)}: Troppo rigide. Con 21+ variabili le combinazioni sono troppe da gestire a mano.
                \item \textbf{Approccio Probabilistico}: Calcola la percentuale di rischio abbandono per ogni cliente: $P(\text{Churn} \mid \text{Dati})$.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{2mm}
    \centering
    \colorbox{LightBg}{\parbox{0.95\textwidth}{\centering
        \small \color{DarkText} \textbf{Conclusione}: Le scelte dei clienti sono imprevedibili. La probabilità è indispensabile per calcolare il rischio reale di abbandono nel CRM.
    }}

    \note{Come si passa dal PEAS al Machine Learning? Nella Slide 2 abbiamo definito l'Agente Razionale in termini di PEAS: sensori, attuatori, ambiente CRM e performance. Ma come fa l'agente a trasformare quello che legge dai sensori nelle decisioni degli attuatori? In un ambiente reale non può usare semplici regole fisse: le combinazioni di clienti sono troppe e il comportamento umano è imprevedibile. Impara quindi dai dati del passato per calcolare una percentuale di rischio di abbandono P(Churn|Dati).}
\end{frame}

% SLIDE 5: CLUSTERING K-MEANS & SELEZIONE K
% ==========================================
\begin{frame}{Apprendimento Non Supervisionato: Clustering K-Means}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \textbf{Scopo del Clustering e Scelta di K}
            \vspace{1.5mm}
            \begin{itemize}
                \small
                \setlength{\itemsep}{3pt}
                \item \textbf{Obiettivo}: Raggruppare spontaneamente i clienti in segmenti omogenei senza usare l'etichetta Churn.
                \item \textbf{Scelta del numero di gruppi ($K$)}:
                \begin{itemize}
                    \small
                    \setlength{\itemsep}{2pt}
                    \item \textbf{$K=2$}: Divisione troppo banale (solo clienti con/senza internet).
                    \item \textbf{$K=3$ (Scelta Ottimale)}: Perfetto equilibrio tra separazione geometrica e utilità commerciale.
                    \item \textbf{$K \ge 4$}: I gruppi iniziano a sovrapporsi e la qualità crolla.
                \end{itemize}
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\textwidth,height=0.44\textheight,keepaspectratio]{images/clustering_scelta_k.png}
            
            \vspace{1.5mm}
            \colorbox{LightBg}{\parbox{\textwidth}{
                \small \color{DarkText}
                \textbf{Come leggere il grafico a destra}:
                \begin{itemize}
                    \small
                    \setlength{\itemsep}{1pt}
                    \item \textbf{Inerzia (Curva Blu)}: Errore interno. Mostra un evidente punto di "gomito" a \textbf{$K=3$}.
                    \item \textbf{Silhouette (Curva Arancione)}: Misura la separazione dei gruppi. \textbf{$K=3$} ($s=0.286$) è la scelta migliore.
                \end{itemize}
            }}
        \end{column}
    \end{columns}

    \note{Nella Slide 5 passiamo all'apprendimento non supervisionato usando K-Means per raggruppare i clienti in segmenti senza usare le etichette di Churn. Per scegliere il numero ottimale di cluster K abbiamo usato due metriche: il Metodo del Gomito sull'Inerzia (curva blu) e lo Silhouette Score (curva arancione). Anche se K=2 ha uno score leggermente alto, crea una divisione banale tra chi ha internet e chi no. Abbiamo scelto K=3 perché rappresenta il punto di gomito ideale e distingue 3 profili di business chiari. Per K maggiore o uguale a 4 la silhouette crolla perché i gruppi si sovrappongono.}
\end{frame}
